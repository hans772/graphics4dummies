188 // Pass 1 (post-order): provisional x for every node.
189 void modify(node* n, int depth, structure_data& d) {
190 if (!n) return;
191 if (depth == (int)d.modifiers.size()) {
192 d.modifiers.push_back(0);
193 d.next_pos.push_back(1);
194 }
195
196 modify(n->left, depth + 1, d);
197 modify(n->right, depth + 1, d);
198
199 const bool leaf = !n->left && !n->right;
200 int ideal;
201 if (n->left && n->right) ideal = (n->left->position.x + n->right->position.x) / 2;
202 else if (n->left) ideal = n->left->position.x + 1;
203 else if (n->right) ideal = n->right->position.x - 1;
204 else ideal = d.next_pos[depth];
205
206 int& mod = d.modifiers[depth];
207 mod = std::max(mod, d.next_pos[depth] - ideal);
208
209 n->position.x = leaf ? ideal : ideal + mod;
210 n->modifier = mod;
211 d.next_pos[depth] = n->position.x + 2;
212 }
213
214 // Pass 2 (pre-order): apply the accumulated shifts and assign y.
215 void place(node* n, int depth, int shift) {
216 if (!n) return;
217 n->position.x += shift;
218 n->position.y = 2 * depth + 1;
219 min_x = std::min(min_x, n->position.x);
220 place(n->left, depth + 1, shift + n->modifier);
221 place(n->right, depth + 1, shift + n->modifier);
222 }

224 void draw_edges(frame& f, const node* n) {
225 const glm::dvec2 a = glm::dvec2(to_pixel(*n, f));
226 for (const node* c : {n->left, n->right}) {
227 if (!c) continue;
228 const glm::dvec2 b = glm::dvec2(to_pixel(*c, f));
229 const glm::dvec2 dir = glm::normalize(b - a); //direction to child
230 const glm::dvec2 p = a + dir * (double)config.radius;
231 const glm::dvec2 q = b - dir * (double)config.radius;
232 f.draw_line_coh_sut((int)std::lround(p.x), (int)std::lround(p.y),
233 (int)std::lround(q.x), (int)std::lround(q.y), config.edge_color);
234 draw_edges(f, c);
235 }

245 static node* remove(node* n, const T& value, bool& removed) {
246 if (!n) return nullptr;
247 if (value < n->value) {
248 n->left = remove(n->left, value, removed);
249 } else if (n->value < value) {
250 n->right = remove(n->right, value, removed);
251 } else {
252 removed = true;
253 if (!n->left) { node* r = n->right; delete n; return r; }
254 if (!n->right) { node* l = n->left; delete n; return l; }
255 node* s = n->right;
256 while (s->left) s = s->left;
257 n->value = s->value;
258 bool ignored = false;
259 n->right = remove(n->right, n->value, ignored);
260
261 }
262 return n;
263 }

265 static void destroy(node* root) {
266 std::vector<node*> stack;
267 if (root) stack.push_back(root);
268 while (!stack.empty()) {
269 node* n = stack.back();
270 stack.pop_back();
271 if (n->left) stack.push_back(n->left);
272 if (n->right) stack.push_back(n->right);
273 delete n;
274 }
