101 // ------------------------------------------------------------------ rotations
102 //
103 // fix_left_heavy / fix_right_heavy are called when one side of `n` is two levels taller
104 // than the other. They rotate, update the balance factors and return the new subtree
105 // root. `decreased` reports whether the subtree got shorter overall: always true for the
106 // cases insertion can produce, false only for the deletion case where the taller child
107 // is itself balanced (a single rotation that leaves the height unchanged).
108
109 // left subtree of n is too tall
110 static node* fix_left_heavy(node* n, bool& decreased) {
111 node* l = n->left;
112 decreased = l->balance != 0;
113
114 if (l->balance <= 0) { // left-left (or the balanced-child deletion case): single right rotation
115 n->left = l->right;
116 l->right = n;
117 if (l->balance == 0) { n->balance = -1; l->balance = +1; }
118 else { n->balance = 0; l->balance = 0; }
119 return l;
120 }
121
122 node* lr = l->right; // left-right: rotate l left, then n right
123 l->right = lr->left;
124 lr->left = l;
125 n->left = lr->right;
126 lr->right = n;
127 n->balance = lr->balance == -1 ? +1 : 0;
128 l->balance = lr->balance == +1 ? -1 : 0;
129 lr->balance = 0;

131 }

157 // ------------------------------------------------------------------ insertion
158 //
159 // Returns the new root of the subtree. `grown` is set when the subtree got one level
160 // taller; the caller uses it to update its own balance factor.
161 static node* insert(node* n, T& value, bool& grown, bool& inserted) {
162 if (!n) {
163 grown = inserted = true;
164 return new node{std::move(value)};
165 }
166
167 if (value < n->value) {
168 n->left = insert(n->left, value, grown, inserted);
169 if (grown) {
170 if (n->balance > 0) { n->balance = 0; grown = false; } // was right-heavy: now level
171 else if (n->balance == 0) { n->balance = -1; } // now left-heavy, still taller
172 else { bool unused; n = fix_left_heavy(n, unused); grown = false; }
173 }
174 } else if (n->value < value) {
175 n->right = insert(n->right, value, grown, inserted);
176 if (grown) {
177 if (n->balance < 0) { n->balance = 0; grown = false; }
178 else if (n->balance == 0) { n->balance = +1; }
179 else { bool unused; n = fix_right_heavy(n, unused); grown = false; }
180 }
181 } else {
182 grown = false; // duplicate
183 }
184 return n;
185 }

191 static node* left_shrunk(node* n, bool& shrunk) { // n's left subtree became shorter
192 if (n->balance < 0) { n->balance = 0; } // was left-heavy: level now, but shorter
193 else if (n->balance == 0) { n->balance = +1; shrunk = false; } // height unchanged
194 else { n = fix_right_heavy(n, shrunk); }

196 }

198 static node* right_shrunk(node* n, bool& shrunk) { // n's right subtree became shorter
199 if (n->balance > 0) { n->balance = 0; }
200 else if (n->balance == 0) { n->balance = -1; shrunk = false; }
201 else { n = fix_left_heavy(n, shrunk); }
202 return n;
203 }

205 static node* remove(node* n, const T& value, bool& shrunk, bool& removed) {
206 if (!n) {
207 shrunk = false;
208 return nullptr;
209 }
210
211 if (value < n->value) {
212 n->left = remove(n->left, value, shrunk, removed);
213 if (shrunk) n = left_shrunk(n, shrunk);
214 } else if (n->value < value) {
215 n->right = remove(n->right, value, shrunk, removed);
216 if (shrunk) n = right_shrunk(n, shrunk);
217 } else {
218 removed = true;
219 if (!n->left || !n->right) { // zero or one child: splice the node out
220 node* child = n->left ? n->left : n->right;
221 delete n;
222 shrunk = true;
223 return child;
224 }
225 node* s = n->right; // two children: take the in-order successor's value,
226 while (s->left) s = s->left; // then remove that successor from the right subtree
227 n->value = s->value;
228 bool ignored = false;
229 n->right = remove(n->right, n->value, shrunk, ignored);
230 if (shrunk) n = right_shrunk(n, shrunk);
231 }
232 return n;
233 }
